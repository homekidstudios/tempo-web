    }
    VETERINARIO {
        int id_veterinario PK
        string especialidad
    }
    CITA {
        int id_cita PK
        datetime fecha_hora
        int id_mascota FK
        int id_veterinario FK
    }
    TRATAMIENTO {
        int id_tratamiento PK
        string nombre
        float costo
    }
